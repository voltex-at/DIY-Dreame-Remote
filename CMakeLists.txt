cmake_minimum_required(VERSION 3.16)
set(DDR_BOARD "lcdwiki-es3c35p" CACHE STRING "Supported board")
if(NOT DDR_BOARD STREQUAL "lcdwiki-es3c35p")
    message(FATAL_ERROR "Only lcdwiki-es3c35p is supported")
endif()
if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "ES3C35P requires ESP32-S3")
endif()
set(IDF_TARGET "esp32s3" CACHE STRING "" FORCE)
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;boards/lcdwiki-es3c35p/sdkconfig.defaults")
add_compile_definitions(DDR_BOARD_ES3C35P=1)
set(PROJECT_VER "0.1.0-bringup")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(diy_dreame_remote)
